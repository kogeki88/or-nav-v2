// ===== schema.ts =====
// What the omnibar AI may answer: a short reply plus actions for the app.
// Shared by the server (JSON schema in the prompt, answer check) and the app (types + checks).
// The AI never changes data: "draft_*" actions only fill a suggestion the nurse checks and sends for review.

export const PHASE_NAMES = ['Setup', 'Incision', 'Cannulation/Main', 'Closure'] as const;
export const TABS = ['library', 'review', 'me'] as const;

export type OpenItem = { type: 'open_item'; item_id: string };
export type OpenCard = { type: 'open_card'; card_id: string };
export type ShowResults = { type: 'show_results'; item_ids: string[]; card_ids: string[] };
export type DraftItem = {
  type: 'draft_item';
  kind: 'add' | 'edit' | 'remove';
  item_id: string | null;
  name: string | null;
  category: string | null;
  shelf: string | null;
  sub_location: string | null;
  synonyms: string[] | null;
  notes: string | null;
  summary: string;
};
export type DraftCard = {
  type: 'draft_card';
  kind: 'add' | 'edit' | 'remove';
  card_id: string | null;
  surgeon: string | null;
  procedure: string | null;
  specialty: string | null;
  notes: string | null;
  summary: string;
};
export type DraftCardLine = {
  type: 'draft_card_line';
  kind: 'add' | 'edit' | 'remove';
  card_id: string;
  card_item_id: string | null;
  item_id: string | null;
  phase: (typeof PHASE_NAMES)[number] | null;
  quantity: number | null;
  open_or_hold: 'open' | 'hold' | null;
  note: string | null;
  summary: string;
};
export type SwitchDepartment = { type: 'switch_department'; department_id: string };
export type GoTo = { type: 'go_to'; tab: (typeof TABS)[number] };

export type OmniAction = OpenItem | OpenCard | ShowResults | DraftItem | DraftCard | DraftCardLine | SwitchDepartment | GoTo;
export type OmniAnswer = { reply: string; actions: OmniAction[] };

const str = { type: 'string' } as const;
/** A fixed value, written as a one-item enum (not "const"). */
const one = (value: string) => ({ type: 'string', enum: [value] });
const strOrNull = { type: ['string', 'null'] } as const;
const kind = { type: 'string', enum: ['add', 'edit', 'remove'] } as const;

function obj(properties: Record<string, unknown>) {
  return { type: 'object', properties, required: Object.keys(properties), additionalProperties: false };
}

/** JSON schema for the answer. It goes into the prompt (JSON_RULES), and answer.ts checks every answer against it. */
export const ANSWER_SCHEMA = obj({
  reply: { ...str, description: 'Short plain answer for the nurse, 1-3 sentences.' },
  actions: {
    type: 'array',
    items: {
      anyOf: [
        obj({ type: one('open_item'), item_id: str }),
        obj({ type: one('open_card'), card_id: str }),
        obj({ type: one('show_results'), item_ids: { type: 'array', items: str }, card_ids: { type: 'array', items: str } }),
        obj({
          type: one('draft_item'),
          kind,
          item_id: strOrNull,
          name: strOrNull,
          category: strOrNull,
          shelf: strOrNull,
          sub_location: strOrNull,
          synonyms: { type: ['array', 'null'], items: str },
          notes: strOrNull,
          summary: str,
        }),
        obj({
          type: one('draft_card'),
          kind,
          card_id: strOrNull,
          surgeon: strOrNull,
          procedure: strOrNull,
          specialty: strOrNull,
          notes: strOrNull,
          summary: str,
        }),
        obj({
          type: one('draft_card_line'),
          kind,
          card_id: str,
          card_item_id: strOrNull,
          item_id: strOrNull,
          phase: { type: ['string', 'null'], enum: [...PHASE_NAMES, null] },
          quantity: { type: ['integer', 'null'] },
          open_or_hold: { type: ['string', 'null'], enum: ['open', 'hold', null] },
          note: strOrNull,
          summary: str,
        }),
        obj({ type: one('switch_department'), department_id: str }),
        obj({ type: one('go_to'), tab: { type: 'string', enum: [...TABS] } }),
      ],
    },
  },
});

/** One turn of the chat, as the app sends it (already scrubbed). */
export type OmniTurn = { role: 'user' | 'assistant'; text: string };

/** The shape of a database id (department, card). */
export const UUID = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i;

/** The case the nurse is working on. */
export type OmniCase = { surgeon: string | null; procedure: string | null; card_id: string | null };
/** The request body the app sends to the omnibar function. case: optional; old app versions do not send it. */
export type OmniRequest = { message: string; history: OmniTurn[]; department_id: string; case?: OmniCase };

// ===== scrub.ts =====
// Patient-information scrubber for the omnibar. Pure TypeScript: the app runs it before anything is sent,
// and the server (supabase/functions/omnibar) runs the same file again.
//
// Based on the HIPAA Safe Harbor identifiers that can appear in typed text. It replaces identifiers
// (dates, numbers, contacts, addresses) and BLOCKS the message when it sees a likely patient name or a
// date of birth, because names cannot be removed reliably. It is a backstop, not a guarantee:
// staff must still not type patient information.

export type ScrubResult =
  | { ok: true; text: string; removed: string[] }
  | { ok: false; reason: string };

type Rule = { label: string; tag: string; re: RegExp };

const MONTHS = 'jan(?:uary)?|feb(?:ruary)?|mar(?:ch)?|apr(?:il)?|may|jun(?:e)?|jul(?:y)?|aug(?:ust)?|sep(?:t(?:ember)?)?|oct(?:ober)?|nov(?:ember)?|dec(?:ember)?';
const UNITS = 'mm|cm|ml|mL|cc|mg|mcg|g|kg|l|in|fr|french|gauge|ga|units?|u';

// Order matters: specific patterns first, so a phone number is not eaten by the long-number rule.
const RULES: Rule[] = [
  { label: 'an email address', tag: '[EMAIL]', re: /\b[\w.+-]+@[\w-]+(?:\.[\w-]+)+\b/gi },
  { label: 'a web address', tag: '[URL]', re: /\b(?:https?:\/\/|www\.)\S+/gi },
  { label: 'an IP address', tag: '[IP]', re: /\b(?:\d{1,3}\.){3}\d{1,3}\b/g },
  { label: 'a Social Security number', tag: '[SSN]', re: /\b\d{3}-\d{2}-\d{4}\b/g },
  { label: 'a phone number', tag: '[PHONE]', re: /(?:\+?1[\s.-]?)?\(?\b\d{3}\)?[\s.-]\d{3}[\s.-]\d{4}\b/g },
  { label: 'a date', tag: '[DATE]', re: /\b\d{1,2}[/-]\d{1,2}[/-]\d{2,4}\b/g },
  { label: 'a date', tag: '[DATE]', re: /\b\d{4}-\d{1,2}-\d{1,2}\b/g },
  { label: 'a date', tag: '[DATE]', re: new RegExp(`\\b(?:${MONTHS})\\.?\\s+\\d{1,2}(?:st|nd|rd|th)?(?:,?\\s+\\d{4})?\\b`, 'gi') },
  { label: 'a date', tag: '[DATE]', re: new RegExp(`\\b\\d{1,2}(?:st|nd|rd|th)?\\s+(?:of\\s+)?(?:${MONTHS})\\b\\.?(?:,?\\s+\\d{4})?`, 'gi') },
  { label: 'an age over 89', tag: '[AGE]', re: /\b(?:9\d|1\d\d)\s*(?:-?\s*(?:y\/?o|yo|yrs?|years?)(?:\s*old)?)\b/gi },
  {
    label: 'a street address',
    tag: '[ADDRESS]',
    re: /\b\d{1,6}\s+(?:[A-Za-z]+\s+){1,3}(?:st|street|ave|avenue|rd|road|blvd|boulevard|drive|ln|lane|ct|court|way|pl|place|hwy|highway)\b\.?/gi,
  },
  {
    label: 'a record or account number',
    tag: '[ID]',
    re: /\b(?:mrn|med(?:ical)?\s*rec(?:ord)?s?|acct|account|fin|csn|encounter|visit|chart|case|serial|s\/n|insurance|policy|member|license|plate)\s*(?:no\.?|number|num|#)?\s*[:#]?\s*[a-z]*\d[\w-]{2,}/gi,
  },
  { label: 'a long number', tag: '[ID]', re: new RegExp(`\\b\\d{5,}\\b(?!\\s*(?:${UNITS})\\b)`, 'g') },
];

// A date of birth almost always comes with a name, so it blocks the whole message.
const DOB = /\b(?:dob|d\.o\.b\.?|date\s+of\s+birth|birth\s*date|birthday|born\s+(?:on|in))\b/i;

// Words that often follow "patient" or "room 4" and are not names.
const NOT_NAMES = new Set(
  (
    'a an the is was has had in on at for to of and or with without who that this needs need ready arrived arriving ' +
    'positioning position positioned prep prepped care safety room bed side weight height allergy allergies allergic ' +
    'supine prone lateral lithotomy sitting beach chair trendelenburg reverse transfer transport consent family ' +
    'count counts warming warmer pad roll rolls draped drape will can should may if when after before ' +
    'pre post op preop postop intra or pacu holding asleep awake intubated extubated stable unstable ' +
    'education teaching identifiers identification name names id band'
  ).split(' '),
);

// Common US first names and surnames. Checked ONLY right after a patient cue ("pt maria", "patient lopez"),
// because many are also instrument names (Kelly, Mayo, Allis) that are fine elsewhere.
const COMMON_NAMES = new Set(

    'jessica thomas sarah charles karen christopher lisa daniel nancy matthew betty anthony margaret mark sandra ' +
    'donald ashley steven kimberly paul emily andrew donna joshua michelle kenneth carol kevin amanda brian dorothy ' +
    'george melissa timothy deborah ronald stephanie edward rebecca jason sharon jeffrey laura ryan cynthia jacob ' +
    'kathleen gary amy nicholas angela eric shirley jonathan anna stephen brenda larry pamela justin emma scott ' +
    'nicole brandon helen benjamin samantha samuel katherine gregory christine alexander debra frank rachel patrick ' +
    'carolyn raymond janet jack catherine dennis maria jerry heather tyler diane aaron ruth jose julie adam olivia ' +
    'henry joyce nathan virginia douglas victoria zachary kelly peter lauren kyle christina juan joan carlos luis ' +
    'jorge miguel ana rosa carmen sofia lucia mohammed ahmed wei li mei hiroshi yuki priya raj ' +
    'smith johnson williams brown jones garcia miller davis rodriguez martinez hernandez lopez gonzalez wilson ' +
    'anderson thomas taylor moore jackson martin lee perez thompson white harris sanchez clark ramirez lewis ' +

    'sanders patel myers long ross foster jimenez chen wang zhang liu singh kumar shah khan'
  ).split(' '),
);

const CUE =/\b(?:pt|pts|patient|patients|mr|mrs|ms|miss|mx)\b\.?\s*(?:'s\s+)?(?:name\s*(?:is|:)?\s*)?([A-Za-z][A-Za-z'’-]+)/gi;
const NAME_LABEL = /\b(?:pt|patient)?\s*name\s*(?:is|:)\s*[A-Za-z]/i;
const ROOM = /\b(?:room|rm|bed|bay)\s*#?\s*\d+[a-z]?\s*[-:,]?\s*([A-Z][a-z'’-]+)/g;
const TITLE = /\b(?:mr|mrs|ms|miss|mx)\b\.?\s+([A-Za-z][A-Za-z'’-]+)/gi;

function allowedName(word: string, allow: Set<string>): boolean {
  const w = word.toLowerCase();
  return NOT_NAMES.has(w) || allow.has(w);
}

/**
 * Cleans one message. `allowNames` are staff names that may appear (surgeons in the library).
 * Returns the cleaned text, or ok: false with a plain-language reason to show the nurse.
 */
export function scrub(input: string, allowNames: string[] = []): ScrubResult {
  const allow = new Set(allowNames.flatMap((n) => n.toLowerCase().split(/[^a-z'’-]+/)).filter((w) => w.length > 1));
  const text = input.normalize('NFKC');

  if (DOB.test(text)) return { ok: false, reason: 'Remove the date of birth. Do not type patient information here.' };
  if (NAME_LABEL.test(text)) return { ok: false, reason: "Remove the patient's name. Do not type patient information here." };

  for (const m of text.matchAll(TITLE)) {
    // "Mrs. Garcia", "mr jones"; not "has ms for years".
    if (!allowedName(m[1], allow)) return { ok: false, reason: "Remove the patient's name. Do not type patient information here." };
  }
  for (const m of text.matchAll(CUE)) {
    const word = m[1];
    // "patient Smith", "pt Jones": a capitalized word that is not an ordinary word, or a common name in any case.
    const looksLikeName = (/^[A-Z]/.test(word) && !allowedName(word, allow)) || (COMMON_NAMES.has(word.toLowerCase()) && !allow.has(word.toLowerCase()));
    if (looksLikeName) {
      return { ok: false, reason: "Remove the patient's name. Do not type patient information here." };
    }
  }
  for (const m of text.matchAll(ROOM)) {
    if (!allowedName(m[1], allow)) return { ok: false, reason: "Remove the patient's name. Do not type patient information here." };
  }

  let out = text;
  const removed: string[] = [];
  for (const rule of RULES) {
    out = out.replace(rule.re, () => {
      if (!removed.includes(rule.label)) removed.push(rule.label);
      return rule.tag;
    });
  }
  return { ok: true, text: out, removed };
}

/** "a date and a phone number" — for the note under a sent message. */
export function describeRemoved(removed: string[]): string {
  if (removed.length === 0) return '';
  if (removed.length === 1) return removed[0];
  return `${removed.slice(0, -1).join(', ')} and ${removed[removed.length - 1]}`;
}

// ===== answer.ts =====
// Checks a model's answer text against ANSWER_SCHEMA. Some models cannot be forced into the schema
// (they only get the JSON rules in the prompt), so every answer is checked here. A broken answer is null,
// and the server then tries the next model. Pure TypeScript: no Deno code, so Jest can test it.

type Schema = {
  type?: string | string[];
  enum?: unknown[];
  properties?: Record<string, Schema>;
  required?: string[];
  additionalProperties?: boolean;
  items?: Schema;
  anyOf?: Schema[];
};

function typeOk(t: string, v: unknown): boolean {
  switch (t) {
    case 'null': return v === null;
    case 'string': return typeof v === 'string';
    case 'integer': return typeof v === 'number' && Number.isInteger(v);
    case 'number': return typeof v === 'number' && Number.isFinite(v);
    case 'boolean': return typeof v === 'boolean';
    case 'array': return Array.isArray(v);
    case 'object': return typeof v === 'object' && v !== null && !Array.isArray(v);
    default: return false;
  }
}

/** True when the value fits the schema. Covers only the parts ANSWER_SCHEMA uses. */
function fits(s: Schema, v: unknown): boolean {
  if (s.anyOf) return s.anyOf.some((sub) => fits(sub, v));
  if (s.type !== undefined) {
    const types = Array.isArray(s.type) ? s.type : [s.type];
    if (!types.some((t) => typeOk(t, v))) return false;
  }
  if (s.enum && !s.enum.includes(v)) return false;
  if (Array.isArray(v)) return !s.items || v.every((x) => fits(s.items!, x));
  if (typeof v === 'object' && v !== null) {
    const o = v as Record<string, unknown>;
    const props = s.properties ?? {};
    if ((s.required ?? []).some((k) => !(k in o))) return false;
    if (s.additionalProperties === false && Object.keys(o).some((k) => !(k in props))) return false;
    return Object.entries(o).every(([k, x]) => !props[k] || fits(props[k], x));
  }
  return true;
}

/** The JSON object in the text: the whole text, a ```json fence, or from the first "{" to the last "}". */
function extractJson(text: string): unknown {
  const tries = [text.trim()];
  const fence = /```(?:json)?\s*([\s\S]*?)```/i.exec(text);
  if (fence) tries.push(fence[1].trim());
  const start = text.indexOf('{');
  const end = text.lastIndexOf('}');
  if (start >= 0 && end > start) tries.push(text.slice(start, end + 1));
  for (const t of tries) {
    try {
      return JSON.parse(t);
    } catch {
      // try the next form
    }
  }
  return undefined;
}

/** The prompt shows ids as "item:<id>", "card:<id>", "line:<id>", "dept:<id>"; models often copy the label. */
const ID_LABEL = /^(?:item|card|line|dept):/;
const unlabel = (v: unknown) => (typeof v === 'string' ? v.replace(ID_LABEL, '') : v);

function cleanIds(a: Record<string, unknown>): Record<string, unknown> {
  const out: Record<string, unknown> = {};
  for (const [k, v] of Object.entries(a)) {
    if (k === 'item_ids' || k === 'card_ids') out[k] = (v as unknown[]).map(unlabel);
    else out[k] = k.endsWith('_id') ? unlabel(v) : v;
  }
  return out;
}

/** The answer when the text holds JSON that matches ANSWER_SCHEMA; otherwise null (a failure). */
export function checkAnswer(text: string): OmniAnswer | null {
  const value = extractJson(text);
  if (value === undefined || !fits(ANSWER_SCHEMA as Schema, value)) return null;
  const answer = value as OmniAnswer;
  return { reply: answer.reply, actions: answer.actions.map((a) => cleanIds(a) as OmniAnswer['actions'][number]) };
}

type Row = Record<string, unknown>;

/** A short text or null; undefined means the value is not valid. */
const textOrNull = (v: unknown) => (v === null || v === undefined ? null : typeof v === 'string' && v.length <= 200 ? v : undefined);

/** The optional `case` field. Missing (old app versions) or odd: undefined, and the message is still answered. */
export function readCase(raw: unknown): OmniCase | undefined {
  if (!raw || typeof raw !== 'object') return undefined;
  const c = raw as Record<string, unknown>;
  const surgeon = textOrNull(c.surgeon);
  const procedure = textOrNull(c.procedure);
  if (surgeon === undefined || procedure === undefined) return undefined;
  if (typeof c.card_id !== 'string' || !UUID.test(c.card_id)) return undefined;
  return { surgeon, procedure, card_id: c.card_id };
}

/**
 * One line for the prompt: the case's card and that "his card", "the closure" and so on mean it.
 * Empty when there is no case or its card is not in this department. The words come from the card, not the phone.
 */
export function caseNote(c: OmniCase | null | undefined, cards: Row[]): string {
  if (!c?.card_id) return '';
  const card = cards.find((k) => k.id === c.card_id);
  if (!card) return '';
  return `The nurse is working on this case: card:${card.id} ${card.procedure} | ${card.surgeon}. "His card", "her card", "the card", "the closure", "the setup" and the other phases mean this card, unless the nurse names another one.`;
}

/** The department's library as compact lines for the model (ids included, sorted, no timestamps). */
export function libraryContext(opts: {
  departmentName: string;
  myDepartments: { id: string; label: string }[];
  items: Row[];
  cards: Row[];
  cardItems: Row[];
}): string {
  const byName = (a: Row, b: Row) => String(a.name ?? a.procedure).localeCompare(String(b.name ?? b.procedure));
  const itemName = new Map(opts.items.map((i) => [String(i.id), String(i.name)]));
  const lines: string[] = [];
  lines.push(`Current department: ${opts.departmentName}`);
  lines.push('Your departments:');
  for (const d of [...opts.myDepartments].sort((a, b) => a.label.localeCompare(b.label))) lines.push(`- dept:${d.id} ${d.label}`);

  lines.push('', `Items (${opts.items.length}):`);
  for (const i of [...opts.items].sort(byName)) {
    const syn = Array.isArray(i.synonyms) && i.synonyms.length ? ` | also: ${(i.synonyms as string[]).join(', ')}` : '';
    const where = [i.shelf, i.sub_location].filter(Boolean).join(' / ') || 'no location';
    const cat = i.category ? ` | ${i.category}` : '';
    const notes = i.notes ? ` | note: ${i.notes}` : '';
    lines.push(`- item:${i.id} ${i.name}${cat} | at ${where}${syn}${notes}`);
  }

  lines.push('', `Preference cards (${opts.cards.length}):`);
  for (const c of [...opts.cards].sort(byName)) {
    lines.push(`- card:${c.id} ${c.procedure} | ${c.surgeon}${c.specialty ? ` | ${c.specialty}` : ''}${c.notes ? ` | note: ${c.notes}` : ''}`);
    const mine = opts.cardItems
      .filter((ci) => ci.card_id === c.id)
      .sort((a, b) => String(a.phase).localeCompare(String(b.phase)) || Number(a.sort_order) - Number(b.sort_order));
    for (const ci of mine) {
      lines.push(
        `    line:${ci.id} ${ci.phase}: ${ci.quantity} x ${itemName.get(String(ci.item_id)) ?? 'unknown item'} (item:${ci.item_id}, ${ci.open_or_hold})${ci.note ? ` note: ${ci.note}` : ''}`,
      );
    }
  }
  return lines.join('\n');
}

// ===== core.ts =====
// The AI calls behind the omnibar, all through OpenRouter: the models are tried in order, free first.
// No Deno or Supabase code here, so it can be tested from a PC.

/**
 * The order Sean picked after the ticket #25 test (qa/25/compare-models.html): Nemotron Ultra is free and
 * answered 14 of 14; Claude Haiku is paid (about $0.0007 a message) and answers only when Ultra cannot.
 * The label is what the nurse sees on each answer.
 */
export const MODELS = [
  { id: 'nvidia/nemotron-3-ultra-550b-a55b:free', label: 'Nemotron Ultra' },
  { id: 'anthropic/claude-haiku-5.5', label: 'Claude Haiku' },
] as const;
export type Model = { id: string; label: string };

/** What the nurse reads when no model answered. */
export const BUSY = 'The assistant is busy. Try again in a moment. Search still works.';

const OPENROUTER_URL = 'https://openrouter.ai/api/v1/chat/completions';

/** caseNote: the case line (see prompt.ts), after the library. */
export type AskOpts = { context: string; caseNote?: string; history: OmniTurn[]; message: string };
/** Asks one model; throws when it fails or its answer is broken. */
export type ModelAsk = (modelId: string, opts: AskOpts) => Promise<OmniAnswer>;

/** The OpenRouter request body. No forced schema (Anthropic refuses ours); the JSON rules are in the prompt. */
export function openRouterBody(model: string, opts: AskOpts) {
  return {
    model,
    messages: [
      { role: 'system', content: [SYSTEM_PROMPT, JSON_RULES, opts.context, opts.caseNote].filter(Boolean).join('\n\n') },
      ...opts.history.map((t) => ({ role: t.role, content: t.text })),
      { role: 'user', content: opts.message },
    ],
    max_tokens: 8192, // Nemotron thinks before answering, and thinking counts here
  };
}

/** One model. Throws "status N", a TimeoutError, or "bad answer" (JSON that does not match the schema). */
export async function askOpenRouter(
  apiKey: string,
  model: string,
  opts: AskOpts,
  fetchFn: typeof fetch = fetch,
  timeoutMs = 20_000,
): Promise<OmniAnswer> {
  const res = await fetchFn(OPENROUTER_URL, {
    method: 'POST',
    headers: { Authorization: `Bearer ${apiKey}`, 'Content-Type': 'application/json', 'X-Title': 'OR-Nav' },
    body: JSON.stringify(openRouterBody(model, opts)),
    signal: AbortSignal.timeout(timeoutMs),
  });
  const data = (await res.json().catch(() => ({}))) as {
    choices?: { message?: { content?: string | null } }[];
    error?: { code?: number };
  };
  if (!res.ok) throw new Error(`status ${res.status}`);
  if (data.error) throw new Error(`status ${data.error.code ?? 'error'}`);
  const answer = checkAnswer(data.choices?.[0]?.message?.content ?? '');
  if (!answer) throw new Error('bad answer');
  return answer;
}

/**
 * Tries each model in order; on ANY failure (error, rate limit, timeout, broken answer) the next model gets the
 * same request. Throws BUSY when all fail. `log` receives model names and failure kinds, never text.
 */
export async function answerWithFallback(
  models: readonly Model[],
  ask: ModelAsk,
  opts: AskOpts,
  log: (event: string) => void = () => {},
): Promise<{ answer: OmniAnswer; model: string }> {
  for (const m of models) {
    try {
      const answer = await ask(m.id, opts);
      log(`${m.label} ok`);
      return { answer, model: m.label };
    } catch (e) {
      log(`${m.label} failed (${failureKind(e)})`);
    }
  }
  throw new Error(BUSY);
}

/** A short failure kind for the log: a status number, "timeout", "bad answer" or "error". Never message text. */
function failureKind(e: unknown): string {
  if (e instanceof Error && e.name === 'TimeoutError') return 'timeout';
  const msg = e instanceof Error ? e.message : '';
  if (msg === 'bad answer') return 'bad answer';
  const m = /status (\d{3})/.exec(msg);
  return m ? m[1] : 'error';
}

// ===== index.ts =====
// Supabase Edge Function "omnibar": the only place the AI key is used (one OpenRouter key).
// Steps: signed-in user → member of the department → scrub again → daily limit → load library (as the user)
// → the models in order (core.ts MODELS), free first.
// Privacy: never log message text. Logs hold only event names and counts.
import { createClient } from 'npm:@supabase/supabase-js@2.119.0';


const DAILY_LIMIT = Number(Deno.env.get('OMNIBAR_DAILY_LIMIT') ?? '50');
const MAX_MESSAGE = 1000;
const MAX_HISTORY = 10;

const json = (status: number, body: unknown) =>
  new Response(JSON.stringify(body), { status, headers: { 'Content-Type': 'application/json' } });
const fail = (status: number, error: string) => json(status, { error });

/** The project's public key: the new publishable key ("default"), or the legacy anon key. Never the secret key. */
function publicKey(): string {
  try {
    const keys = JSON.parse(Deno.env.get('SUPABASE_PUBLISHABLE_KEYS') ?? '{}') as Record<string, string>;
    if (keys.default) return keys.default;
  } catch {
    // fall through to the legacy key
  }
  return Deno.env.get('SUPABASE_ANON_KEY') ?? '';
}

function readBody(raw: unknown): OmniRequest | null {
  if (!raw || typeof raw !== 'object') return null;
  const b = raw as Record<string, unknown>;
  if (typeof b.message !== 'string' || !b.message.trim() || b.message.length > MAX_MESSAGE) return null;
  if (typeof b.department_id !== 'string' || !UUID.test(b.department_id)) return null;
  const history = Array.isArray(b.history) ? b.history : [];
  const turns: OmniTurn[] = [];
  for (const t of history.slice(-MAX_HISTORY)) {
    if (t && (t.role === 'user' || t.role === 'assistant') && typeof t.text === 'string' && t.text.length <= 4000) {
      turns.push({ role: t.role, text: t.text });
    }
  }
  // `case` is optional: old app versions do not send it, and an odd one is dropped, not refused.
  return { message: b.message, history: turns, department_id: b.department_id, case: readCase(b.case) };
}

Deno.serve(async (req) => {
  if (req.method !== 'POST') return fail(405, 'Use POST.');
  const auth = req.headers.get('Authorization');
  if (!auth) return fail(401, 'Sign in first.');

  const apiKey = Deno.env.get('OPENROUTER_API_KEY');
  if (!apiKey) {
    console.error('omnibar: no AI key is set');
    return fail(503, 'The assistant is not set up yet.');
  }

  // Acts as the signed-in user, so row-level security limits what can be read.
  const supabase = createClient(Deno.env.get('SUPABASE_URL')!, publicKey(), {
    global: { headers: { Authorization: auth } },
    auth: { persistSession: false },
  });
  const { data: userData, error: userError } = await supabase.auth.getUser(auth.replace(/^Bearer\s+/i, ''));
  if (userError || !userData.user) return fail(401, 'Sign in again.');
  const userId = userData.user.id;

  let body: OmniRequest | null = null;
  try {
    body = readBody(await req.json());
  } catch {
    body = null;
  }
  if (!body) return fail(400, 'That message could not be read.');

  const { data: memberships, error: memError } = await supabase
    .from('memberships')
    .select('department_id, departments(name, hospitals(name))')
    .eq('user_id', userId);
  if (memError) return fail(500, 'Could not check your departments.');
  type MembershipRow = { department_id: string; departments: { name: string; hospitals: { name: string } | null } | null };
  const mine = (memberships ?? []) as unknown as MembershipRow[];
  const current = mine.find((m) => m.department_id === body!.department_id);
  if (!current) return fail(403, "You're not a member of that department.");

  const [items, cards, cardItems] = await Promise.all([
    supabase.from('items').select('id,name,category,shelf,sub_location,synonyms,notes').eq('department_id', body.department_id),
    supabase.from('preference_cards').select('id,surgeon,procedure,specialty,notes').eq('department_id', body.department_id),
    supabase.from('card_items').select('id,card_id,item_id,phase,quantity,open_or_hold,note,sort_order').eq('department_id', body.department_id),
  ]);
  if (items.error || cards.error || cardItems.error) return fail(500, 'Could not load the library.');

  // Second scrub, with the same rules as the phone. Staff (surgeon) names are allowed.
  const staff = (cards.data ?? []).map((c) => String(c.surgeon));
  const clean = scrub(body.message, staff);
  if (!clean.ok) {
    console.log('omnibar: blocked by scrubber');
    return fail(422, clean.reason);
  }
  const history: OmniTurn[] = [];
  for (const t of body.history) {
    if (t.role === 'assistant') {
      history.push(t);
      continue;
    }
    const s = scrub(t.text, staff);
    if (s.ok) history.push({ role: 'user', text: s.text });
  }
  // Some models need turns that alternate user/assistant, starting with the user.
  while (history.length && history[0].role !== 'user') history.shift();
  const alternating = history.filter((t, i) => i === 0 || t.role !== history[i - 1].role);
  if (alternating.length && alternating[alternating.length - 1].role === 'user') alternating.pop();

  const { error: quotaError } = await supabase.rpc('use_ai_quota', { p_limit: DAILY_LIMIT });
  if (quotaError) {
    if (/daily limit/i.test(quotaError.message)) return fail(429, `You've used today's ${DAILY_LIMIT} assistant messages. Search still works.`);
    return fail(500, 'Could not check the daily limit.');
  }

  const label = (m: MembershipRow) => [m.departments?.hospitals?.name, m.departments?.name].filter(Boolean).join(' · ');
  const context = libraryContext({
    departmentName: label(current),
    myDepartments: mine.map((m) => ({ id: m.department_id, label: label(m) })),
    items: items.data ?? [],
    cards: cards.data ?? [],
    cardItems: cardItems.data ?? [],
  });

  try {
    // The case's card must be one of this department's cards; any other card_id is ignored.
    const note = caseNote(body.case, cards.data ?? []);
    const opts = { context, caseNote: note, history: alternating, message: clean.text };
    // Each model fails fast (20 s), so the next one still answers within a nurse's patience.
    const ask: ModelAsk = (modelId, o) => askOpenRouter(apiKey, modelId, o);
    const { answer, model } = await answerWithFallback(MODELS, ask, opts, (event) => console.log(`omnibar: ${event}`));
    console.log(`omnibar: ${model}, ${answer.actions.length} actions`);
    return json(200, { ...answer, model, removed: clean.removed });
  } catch {
    return fail(503, BUSY);
  }
});
